<!-- Script Logika Filter Kategori -->
<script>
    $(document).ready(function(){
        $('.filter-btn').click(function(){
            // Reset state tombol
            $('.filter-btn').removeClass('active');
            $(this).addClass('active');
            
            // Ambil data filter
            var filter = $(this).attr('data-filter');
            
            // Logika sembunyikan & tampilkan kartu
            if(filter == 'all'){
                $('.item-card').fadeIn(500);
            } else {
                $('.item-card').hide();
                $('.item-card.' + filter).fadeIn(500);
            }
        });
    });
</script>
